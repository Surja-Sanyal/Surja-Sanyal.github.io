%% Check for published articles: skips if note is "(Under Review)" OR "(Under Revision)"
\defbibcheck{published}{%
  \iffieldequalstr{note}{(Under Review)}
    {\skipentry}
    {\iffieldequalstr{note}{(Under Revision)}
       {\skipentry}
       {}% Keeps the entry if it is neither
    }%
}

%% Check for manuscripts in progress: keeps if note is "(Under Review)" OR "(Under Revision)"
\defbibcheck{inreview}{%
  \iffieldequalstr{note}{(Under Review)}
    {} % Keeps the entry
    {\iffieldequalstr{note}{(Under Revision)}
       {} % Keeps the entry
       {\skipentry}% Skips if it is neither
    }%
}
